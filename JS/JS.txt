function filterMenu() {
  // Get the value typed into the search box
  let input = document.getElementById('searchInput').value.toLowerCase();
  // Get all menu items
  let items = document.getElementsByClassName('menu-item');

  // Loop through all list items, and hide those who don't match the search query
  for (let i = 0; i < items.length; i++) {
    let txtValue = items[i].textContent || items[i].innerText;
    if (txtValue.toLowerCase().indexOf(input) > -1) {
      items[i].style.display = "";
    } else {
      items[i].style.display = "none";
    }
  }
}

var acc = document.getElementsByClassName("accordion");
for (let i = 0; i < acc.length; i++) {
  acc[i].addEventListener("click", function() {
    this.classList.toggle("active");
    var panel = this.nextElementSibling;
    if (panel.style.display === "block") {
      panel.style.display = "none";
    } else {
      panel.style.display = "block";
    }
  });
}

// --- ENQUIRY FORM VALIDATION & AJAX ---
document.getElementById('enquiryForm').addEventListener('submit', function(e) {
  e.preventDefault(); // Prevent page reload
  
  let guests = document.getElementById('guests').value;
  let errorMsg = document.getElementById('enquiryError');
  let successMsg = document.getElementById('enquirySuccess');

  // Validation: Check if guests are more than 0
  if (guests < 1 || guests === "") {
    errorMsg.textContent = "Please enter a valid number of guests.";
    successMsg.textContent = "";
    return; // Stop submission
  }

  errorMsg.textContent = "";
  
  // Simulate AJAX Submission (Since we don't have a real backend)
  // In a real scenario, you would use fetch() to send data to a server.
  successMsg.textContent = `Thank you! We have received your enquiry for ${guests} guests and will email you a quote shortly.`;
  this.reset(); // Clear the form
});

// --- CONTACT FORM VALIDATION & AJAX ---
document.getElementById('contactForm').addEventListener('submit', function(e) {
  e.preventDefault();
  
  let email = document.getElementById('contactEmail').value;
  let errorMsg = document.getElementById('contactError');
  let successMsg = document.getElementById('contactSuccess');

  // Simple validation logic (HTML5 handles basic email format, but we double-check)
  if (!email.includes("@")) {
    errorMsg.textContent = "Please enter a valid email address.";
    successMsg.textContent = "";
    return;
  }

  errorMsg.textContent = "";
  
  // Simulate AJAX
  successMsg.textContent = "Message sent successfully! We will get back to you soon.";
  this.reset();
});
